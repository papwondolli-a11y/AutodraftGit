# Resources
# translations.qrc embeds translations/autodraft_en.qm at :/i18n/ (English UI).
# Regenerate the .qm with translations/build_translations.bat after changing UI strings.
set(RESOURCES
    icons.qrc
    translations.qrc
    help.qrc
)

# Compile definitions
# Debug 빌드에서만 소스 경로 매크로를 정의한다.
# Release(배포) 빌드에서는 APP_SOURCE_PATH가 정의되지 않으므로
# 각 모듈이 QCoreApplication::applicationDirPath()를 사용하게 된다.
add_compile_definitions($<$<CONFIG:Debug>:APP_SOURCE_PATH="${CMAKE_CURRENT_SOURCE_DIR}">)

# Executable target
if(ANDROID)
    add_library(autodraft SHARED
        ${SOURCES}
        ${HEADERS}
        ${RESOURCES}
    )
else()
    add_executable(autodraft
        ${SOURCES}
        ${HEADERS}
        ${RESOURCES}
    )
endif()

# Linking
target_link_libraries(autodraft PRIVATE
    Qt${QT_VERSION_MAJOR}::Core
    Qt${QT_VERSION_MAJOR}::Gui
    Qt${QT_VERSION_MAJOR}::Widgets
    Qt${QT_VERSION_MAJOR}::Qml
    Qt${QT_VERSION_MAJOR}::Sql
    Qt${QT_VERSION_MAJOR}::Network
    Qt${QT_VERSION_MAJOR}::Quick
    Qt${QT_VERSION_MAJOR}::PrintSupport
    Qt${QT_VERSION_MAJOR}::OpenGLWidgets
)

# Install rules (simplified from .pro)
include(GNUInstallDirs)
install(TARGETS autodraft
    BUNDLE DESTINATION .
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
